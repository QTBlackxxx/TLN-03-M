!
{# --- 2. RUTEO ADICIONAL PARA DAR ACCESO A INTERNET A LA NUEVA SEDE --- #}

ip route {{ pool_destino }} {{ mask_pool }} {{ ip_wan_destino }} tag {{ id_tag }}
!
route-map {{ nombre_route }} permit 10
 match tag {{ id_tag }}
!
router bgp {{ id_bgp }}
 address-family ipv4
 redistribute static route-map {{ nombre_route }}
!
